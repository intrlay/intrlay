cmake_minimum_required(VERSION 3.20)
project(intrlay_collection_consumer LANGUAGES CXX)
find_package(intrlay CONFIG REQUIRED)
add_executable(collections main.cpp)
target_link_libraries(collections PRIVATE intrlay::intrlay)
target_include_directories(collections PRIVATE "${INTRLAY_FIXTURES}")
if(MSVC)
    target_compile_options(collections PRIVATE /W4 /WX)
else()
    target_compile_options(collections PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME collections COMMAND collections "${INTRLAY_FIXTURES}")
