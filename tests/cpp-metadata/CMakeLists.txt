cmake_minimum_required(VERSION 3.20)
project(intrlay_metadata_consumer LANGUAGES CXX)
find_package(intrlay CONFIG REQUIRED)
add_executable(metadata main.cpp linkage.cpp)
target_compile_definitions(metadata PRIVATE INTRLAY_CHECK_LINKAGE)
target_link_libraries(metadata PRIVATE intrlay::intrlay)
target_include_directories(metadata PRIVATE "${INTRLAY_FIXTURES}")
if(MSVC)
    target_compile_options(metadata PRIVATE /W4 /WX)
else()
    target_compile_options(metadata PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME metadata COMMAND metadata "${INTRLAY_FIXTURES}")

# Exercise adapter macros and nested sources through the installed headers too.
add_executable(adapters ../cpp/adapters.cpp)
target_link_libraries(adapters PRIVATE intrlay::intrlay)
target_include_directories(adapters PRIVATE "${INTRLAY_FIXTURES}/../regressions")
if(MSVC)
    target_compile_options(adapters PRIVATE /W4 /WX)
else()
    target_compile_options(adapters PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME adapters COMMAND adapters)
